# Size of the offhand icon in pixels (it is normalised to this size).
# Big on purpose: it is meant to look like the item held in the left hand,
# mirroring the engine's main-hand view.
offhand_screen_icon_size (Offhand icon size in pixels) int 380 8 1024

# Pixels of dark slot background around the icon (only if background enabled).
offhand_screen_bg_padding (Background padding around the icon) int 6 0 64

# Anchor position as a FRACTION of the screen (0..1), like all HUD positions.
# Defaults put the icon in the lower-left corner (opposite side of the
# main-hand wield view).
offhand_screen_pos_x (Offhand icon X position, fraction of screen width) float 0.12 0 1
offhand_screen_pos_y (Offhand icon Y position, fraction of screen height) float 0.88 0 1

offhand_screen_show_icon (Show the on-screen offhand icon) bool true

offhand_screen_show_background (Show a dark slot background behind the offhand icon) bool false

offhand_screen_show_count (Show the stack size on the icon) bool true

# Render blocks as 3D cubes (the same software renderer the engine uses for the
# hotbar) instead of using the flat top texture.
offhand_screen_3d_icons (Render blocks as 3D cubes) bool true

# The base "offhand" mod also draws its own small icon next to the hotbar.
# OFF by default so both stay visible; enable to hide the base one.
offhand_screen_hide_base_icon (Hide the base offhand mod's icon next to the hotbar) bool false
