# Size of the offhand icon in HUD pixels (independent of the texture pack
# resolution; icons are normalised to this size before being drawn).
offhand_screen_icon_size (Offhand icon size in pixels) int 64 8 512

# Pixels of dark slot background around the icon.
offhand_screen_bg_padding (Background padding around the icon) int 6 0 64

# Position, relative to the bottom centre of the screen (like the hotbar).
offhand_screen_offset_x (Offhand icon X offset in pixels, relative to hotbar center) int 130
offhand_screen_offset_y (Offhand icon Y offset in pixels, relative to screen bottom) int -110

offhand_screen_show_background (Show a dark slot background behind the offhand icon) bool true

# Show the stack size in the corner of the icon, like Minecraft does.
offhand_screen_show_count (Show the stack size on the icon) bool true

# Render blocks as 3D cubes (the same software renderer the engine uses for the
# hotbar) instead of using the flat top texture.
offhand_screen_3d_icons (Render blocks as 3D cubes) bool true

# Also make the real 3D item that the "offhand" mod attaches to your left arm
# render in first person (the engine hides attached objects there by default).
# It follows your arm, so it is only inside the view when you look down.
offhand_screen_force_first_person (Also show the 3D item in first person) bool true
